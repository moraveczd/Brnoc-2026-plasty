\frame{
	\frametitle{}
	\vfill
	\textbf{Chemická recyklace plastů}
	\begin{itemize}
		\item Plasty se rozkládají na monomery nebo jiné látky, které lze využít v chemickém průmyslu.
		\item Tento proces je poměrně složitý a vyžaduje:
		\begin{itemize}
			\item působení vysokých teplot (\textit{pyrolýza})
			\item solvolýza, nejčastěji hydrolýza, tzn. rozklad vodou
			\item využití vhodných katalyzátorů
		\end{itemize}
		\item Chemická recyklace umožňuje získat původní monomery, a proto není omezena postupnou degradací polymerních řetězců jako mechanická recyklace.
	\end{itemize}
	\vfill
}

\section{Budoucnost}
\frame{
	\frametitle{}
	\vfill
	\begin{itemize}
		\item Omezení zbytečných jednorázových výrobků.
		\item \textit{Design for Recycling}: výrobky navržené tak, aby se snadno recyklovaly.
		\item Automatizované třídění pomocí AI, kamer a robotických systémů.
		\item Rozvoj chemické recyklace (pyrolýza, depolymerace PET).
		\item Větší podíl recyklovaného materiálu v nových výrobcích.
		\item Biologicky rozložitelné plasty pro vybrané aplikace.
		\item Přechod k cirkulární ekonomice: odpad = surovina.
	\end{itemize}
	\vfill
}

\section{Závěr}
\frame{
	\frametitle{}
	\vfill
	\textbf{Take Home Messages}

	\begin{itemize}
		\item Plasty jsou mimořádně užitečné materiály.
		\item Největším problémem není plast sám, ale způsob jeho používání a likvidace.
		\item Mechanická recyklace má své limity.
		\item Chemická recyklace nabízí nové možnosti.
		\item Mikroplasty a PFAS představují nové environmentální výzvy.
		\item Nejlepší odpad je ten, který vůbec nevznikne.
	\end{itemize}

	\textbf{Nejdůležitější otázkou není „Jak se zbavit plastů?“, ale „Jak používat plasty tak, aby se nestaly odpadem?“}
